foreach(target HalTlsMemoryTest HalTlsMemoryNoPsramTest)
  add_executable(${target} HalTlsMemoryTest.cpp ${REPO_ROOT}/lib/hal/HalTlsMemory.cpp)
  target_include_directories(${target} PRIVATE stubs ${REPO_ROOT}/lib/hal)
  target_compile_features(${target} PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
  endif()
  add_test(NAME ${target} COMMAND ${target})
endforeach()
target_compile_definitions(HalTlsMemoryTest PRIVATE BOARD_HAS_PSRAM=1)
